    import { Injectable } from '@nestjs/common'
    import { Buffer } from 'node:buffer'
    import { jsPDF } from 'jspdf'

    export type KartuPesertaPdfData = {
    nomor_pendaftaran: string
    nis?: string
    nama: string
    nisn: string
    tempat_lahir?: string
    tanggal_lahir?: string
    jenis_kelamin?: string
    no_telepon?: string
    email?: string
    asal_sekolah?: string
    npsn_sekolah_asal?: string
    tahun_lulus?: string | number
    program?: string
    gelombang?: string | number
    status_penerimaan?: string
    tahun_ajaran?: string
    tanggal_diterima?: string
    tanggal_diterbitkan?: string
    }

    export type KartuPesertaPdfAssets = {
    logoBase64?: string
    photoBase64?: string
    }

    type PdfField = {
    label: string
    value?: string | number
    important?: boolean
    }

    const COLOR = {
    black: '#000000',
    white: '#ffffff',
    line: '#111111',
    muted: '#444444',
    lightLine: '#d4d4d4'
    }

    @Injectable()
    export class KartuPesertaPdfService {
    generateKartuPeserta(data: KartuPesertaPdfData, assets: KartuPesertaPdfAssets = {}): Buffer {
        const pdf = new jsPDF({
        orientation: 'portrait',
        unit: 'mm',
        format: 'a4'
        })

        this.drawPage(pdf, data, assets)

        return Buffer.from(pdf.output('arraybuffer'))
    }

    private drawPage(pdf: jsPDF, data: KartuPesertaPdfData, assets: KartuPesertaPdfAssets) {
        const pageWidth = pdf.internal.pageSize.getWidth()
        const marginX = 18
        const contentWidth = pageWidth - marginX * 2
        const printDate = data.tanggal_diterbitkan || this.formatDateId(new Date().toISOString())
        const ttl = [data.tempat_lahir, this.formatDateId(data.tanggal_lahir)]
        .filter(Boolean)
        .join(', ')

        pdf.setFillColor(COLOR.white)
        pdf.rect(0, 0, pageWidth, pdf.internal.pageSize.getHeight(), 'F')

        this.drawLetterhead(pdf, assets, marginX, contentWidth)

        pdf.setFont('times', 'bold')
        pdf.setFontSize(12)
        pdf.setTextColor(COLOR.black)
        pdf.text('KARTU PESERTA PENERIMAAN PESERTA DIDIK BARU', pageWidth / 2, 48, {
        align: 'center'
        })

        pdf.setFont('times', 'normal')
        pdf.setFontSize(10)
        pdf.text(`TAHUN AJARAN ${data.tahun_ajaran || '2026/2027'}`, pageWidth / 2, 54, {
        align: 'center'
        })

        pdf.text(`Nomor: ${this.textOrDash(data.nomor_pendaftaran)}`, pageWidth / 2, 61, {
        align: 'center'
        })

        pdf.setFontSize(10)
        pdf.text(
        `Berdasarkan hasil verifikasi administrasi Penerimaan Peserta Didik Baru Yayasan Mukti Daris Sasmita Cendekia Tahun Ajaran ${data.tahun_ajaran || '2026/2027'}, peserta dengan identitas berikut dinyatakan DITERIMA sebagai peserta didik pada program ${this.textOrDash(data.program)}.`,
        marginX,
        73,
        {
            maxWidth: contentWidth,
            align: 'justify',
            lineHeightFactor: 1.45
        }
        )

        this.drawPhoto(pdf, assets, marginX, 91)

        const fieldX = 62
        let y = 93

        y = this.drawSection(pdf, 'IDENTITAS PESERTA', fieldX, y)
        y = this.drawFields(pdf, fieldX, y, [
        { label: 'Nomor Pendaftaran', value: data.nomor_pendaftaran, important: true },
        { label: 'NIS', value: data.nis || '-', important: true },
        { label: 'Nama Lengkap', value: data.nama },
        { label: 'NISN', value: data.nisn },
        { label: 'Tempat, Tanggal Lahir', value: ttl },
        { label: 'Jenis Kelamin', value: data.jenis_kelamin },
        { label: 'No. HP', value: data.no_telepon },
        { label: 'Email', value: data.email }
        ])

        y += 2
        y = this.drawSection(pdf, 'ASAL SEKOLAH', fieldX, y)
        y = this.drawFields(pdf, fieldX, y, [
        { label: 'Nama Sekolah Asal', value: data.asal_sekolah },
        { label: 'NPSN', value: data.npsn_sekolah_asal },
        { label: 'Tahun Lulus', value: data.tahun_lulus }
        ])

        y += 2
        y = this.drawSection(pdf, 'INFORMASI PENERIMAAN', fieldX, y)
        y = this.drawFields(pdf, fieldX, y, [
        { label: 'Status', value: data.status_penerimaan || 'DITERIMA', important: true },
        { label: 'Program/Paket', value: data.program, important: true },
        { label: 'Gelombang', value: data.gelombang },
        { label: 'Tahun Ajaran', value: data.tahun_ajaran || '2026/2027' },
        { label: 'Tanggal Diterima', value: this.formatDateId(data.tanggal_diterima) }
        ])

        this.drawNotes(pdf, marginX, 216, contentWidth)
        this.drawSignature(pdf, pageWidth, printDate)
    }

    private drawLetterhead(pdf: jsPDF, assets: KartuPesertaPdfAssets, marginX: number, contentWidth: number) {
        if (assets.logoBase64) {
        pdf.addImage(assets.logoBase64, 'PNG', marginX, 12, 18, 18)
        } else {
        pdf.setDrawColor(COLOR.black)
        pdf.rect(marginX, 12, 18, 18)
        pdf.setFont('times', 'bold')
        pdf.setFontSize(8)
        pdf.text('LOGO', marginX + 9, 22, { align: 'center' })
        }

        pdf.setTextColor(COLOR.black)
        pdf.setFont('times', 'bold')
        pdf.setFontSize(14)
        pdf.text('MDS CENDEKIA', marginX + 24, 15)

        pdf.setFont('times', 'normal')
        pdf.setFontSize(9)
        pdf.text('Penerimaan Peserta Didik Baru', marginX + 24, 20)
        pdf.text('Alamat: .................................................................', marginX + 24, 25)
        pdf.text('Email: .....................................  Telp: .....................', marginX + 24, 30)

        pdf.setDrawColor(COLOR.black)
        pdf.setLineWidth(0.5)
        pdf.line(marginX, 36, marginX + contentWidth, 36)
        pdf.setLineWidth(0.2)
        pdf.line(marginX, 37.5, marginX + contentWidth, 37.5)
    }

    private drawPhoto(pdf: jsPDF, assets: KartuPesertaPdfAssets, x: number, y: number) {
        pdf.setDrawColor(COLOR.black)
        pdf.setLineWidth(0.2)
        pdf.rect(x, y, 30, 40)

        if (assets.photoBase64) {
        pdf.addImage(assets.photoBase64, 'JPEG', x, y, 30, 40)
        return
        }

        pdf.setFont('times', 'normal')
        pdf.setFontSize(9)
        pdf.setTextColor(COLOR.muted)
        pdf.text('Foto 3x4', x + 15, y + 21, { align: 'center' })
    }

    private drawSection(pdf: jsPDF, title: string, x: number, y: number) {
        pdf.setFont('times', 'bold')
        pdf.setFontSize(10)
        pdf.setTextColor(COLOR.black)
        pdf.text(title, x, y)
        return y + 7
    }

    private drawFields(pdf: jsPDF, x: number, startY: number, fields: PdfField[]) {
        let y = startY

        fields.forEach((field) => {
        this.drawField(pdf, x, y, field)
        y += 6
        })

        return y
    }

    private drawField(pdf: jsPDF, x: number, y: number, field: PdfField) {
        pdf.setFont('times', 'normal')
        pdf.setFontSize(10)
        pdf.setTextColor(COLOR.black)

        pdf.text(field.label, x, y)
        pdf.text(':', x + 42, y)

        pdf.setFont('times', field.important ? 'bold' : 'normal')
        pdf.text(this.textOrDash(field.value), x + 47, y, {
        maxWidth: 80
        })
    }

    private drawNotes(pdf: jsPDF, x: number, y: number, maxWidth: number) {
        pdf.setFont('times', 'bold')
        pdf.setFontSize(10)
        pdf.setTextColor(COLOR.black)
        pdf.text('Keterangan:', x, y)

        const notes = [
        'Kartu peserta ini merupakan dokumen soft file resmi dari sistem PPDB MDS Cendekia.',
        'Peserta wajib menunjukkan kartu ini kepada petugas saat proses daftar ulang.',
        'Peserta wajib membawa dokumen asli sesuai ketentuan panitia saat daftar ulang.',
        'Apabila ditemukan ketidaksesuaian data, panitia berhak melakukan verifikasi ulang.'
        ]

        pdf.setFont('times', 'normal')
        pdf.setFontSize(9.5)

        let currentY = y + 7
        notes.forEach((note, index) => {
        const lines = pdf.splitTextToSize(`${index + 1}. ${note}`, maxWidth)
        pdf.text(lines, x, currentY)
        currentY += lines.length * 5
        })
    }

    private drawSignature(pdf: jsPDF, pageWidth: number, publishedDate: string) {
        const x = pageWidth - 78
        const y = 252

        pdf.setFont('times', 'normal')
        pdf.setFontSize(10)
        pdf.setTextColor(COLOR.black)
        pdf.text(`Diterbitkan pada ${publishedDate}`, x, y)
        pdf.text('Panitia PPDB', x, y + 7)
        pdf.text('Yayasan Mukti Daris Sasmita Cendekia', x, y + 14)

        pdf.text('(________________________)', x, y + 33)
    }

    private formatDateId(value?: string) {
        if (!value) return '-'

        const date = new Date(value)
        if (Number.isNaN(date.getTime())) return value

        return new Intl.DateTimeFormat('id-ID', {
        day: 'numeric',
        month: 'long',
        year: 'numeric'
        }).format(date)
    }

    private textOrDash(value?: string | number) {
        const text = String(value ?? '').trim()
        return text || '-'
    }
    }
